/**
 * Responsive white center wave used behind the UHQ RELATION logo.
 * It is drawn as a real Shape instead of a fixed rectangular Box so the
 * proportions stay consistent on narrow and wide Android screens.
 */
private class LogoWhiteWaveShape : Shape {
    override fun createOutline(
        size: Size,
        layoutDirection: LayoutDirection,
        density: androidx.compose.ui.unit.Density
    ): androidx.compose.ui.graphics.Outline {
        val w = size.width
        val h = size.height

        val path = Path().apply {
            // Start at the upper-left of the white center panel.
            moveTo(w * 0.08f, 0f)

            // Soft downward curve into the center.
            cubicTo(
                w * 0.20f, h * 0.22f,
                w * 0.32f, h * 0.30f,
                w * 0.50f, h * 0.32f
            )

            // Mirror the curve toward the upper-right.
            cubicTo(
                w * 0.68f, h * 0.30f,
                w * 0.80f, h * 0.22f,
                w * 0.92f, 0f
            )

            // Right side and rounded lower corners.
            lineTo(w, 0f)
            lineTo(w, h * 0.72f)
            cubicTo(w, h * 0.90f, w * 0.90f, h, w * 0.76f, h)
            lineTo(w * 0.24f, h)
            cubicTo(w * 0.10f, h, 0f, h * 0.90f, 0f, h * 0.72f)
            lineTo(0f, 0f)
            close()
        }

        return androidx.compose.ui.graphics.Outline.Generic(path)
    }
}

@OptIn(ExperimentalMaterial3Api::class, ExperimentalFoundationApi::class)
@Composable
fun HomeScreen(navController: NavController, initialTab: String, activity: MainActivity) {
    val auth = FirebaseAuth.getInstance()
    val uid = auth.currentUser?.uid

    var selectedTab by rememberSaveable { mutableStateOf(initialTab) }
    val drawerState = rememberDrawerState(initialValue = DrawerValue.Closed)
    val coroutineScope = rememberCoroutineScope()
    val context = LocalContext.current
    val lifecycleOwner = LocalLifecycleOwner.current

    var showSearchOverlay by remember { mutableStateOf(false) }

    // Existing back behavior preserved.
    BackHandler {
        if (showSearchOverlay) {
            showSearchOverlay = false
        } else {
            (context as? Activity)?.finishAffinity()
        }
    }

    // Existing state/backend variables preserved.
    var banners by remember { mutableStateOf<List<BannerModel>>(emptyList()) }
    var isBannerLoading by remember { mutableStateOf(true) }
    var userName by remember { mutableStateOf(globalUserName.ifEmpty { "..." }) }

    var popupVisible by remember { mutableStateOf(false) }
    var latestAppNotif by remember { mutableStateOf<NotificationModel?>(null) }
    var isFlyingToBell by remember { mutableStateOf(false) }

    var hiddenNotifIds by remember { mutableStateOf(setOf<String>()) }
    var globalNotifCount by remember { mutableIntStateOf(0) }
    var userNotifCount by remember { mutableIntStateOf(0) }

    DisposableEffect(lifecycleOwner) {
        val observer = LifecycleEventObserver { _, event ->
            if (event == Lifecycle.Event.ON_RESUME) {
                hiddenNotifIds = context
                    .getSharedPreferences("UHQ_NOTIFS", Context.MODE_PRIVATE)
                    .getStringSet("hidden_notifs", emptySet()) ?: emptySet()
            }
        }

        lifecycleOwner.lifecycle.addObserver(observer)
        onDispose { lifecycleOwner.lifecycle.removeObserver(observer) }
    }

    AppUpdateChecker()

    // Existing Firestore/backend logic preserved.
    DisposableEffect(uid) {
        val firestoreDb = FirebaseFirestore.getInstance()

        val userListener = uid?.let {
            firestoreDb.collection("Users").document(it)
                .addSnapshotListener { snapshot, error ->
                    if (error == null && snapshot != null && snapshot.exists()) {
                        val fullName = snapshot.getString("name") ?: "User"
                        userName = fullName
                        globalUserName = fullName
                    }
                }
        }

        val bannerListener = firestoreDb.collection("AppBanners")
            .addSnapshotListener { snapshot, _ ->
                if (snapshot != null) {
                    banners = snapshot.documents.mapNotNull {
                        it.toObject(BannerModel::class.java)
                    }
                }
                isBannerLoading = false
            }

        var isGlobalInitial = true

        val globalNotifListener = firestoreDb.collection("AppNotifications")
            .orderBy("timestamp", Query.Direction.DESCENDING)
            .limit(30)
            .addSnapshotListener { snapshot, error ->
                if (error == null && snapshot != null && !snapshot.isEmpty) {
                    globalNotifCount =
                        snapshot.documents.count { !hiddenNotifIds.contains(it.id) }

                    val isNewAddition = snapshot.documentChanges.any {
                        it.type == DocumentChange.Type.ADDED
                    }

                    val newNotif =
                        snapshot.documents[0].toObject(NotificationModel::class.java)

                    if (newNotif != null &&
                        !isGlobalInitial &&
                        isNewAddition &&
                        !popupVisible
                    ) {
                        latestAppNotif = newNotif
                        popupVisible = true
                        isFlyingToBell = false

                        try {
                            RingtoneManager.getRingtone(
                                context,
                                RingtoneManager.getDefaultUri(
                                    RingtoneManager.TYPE_NOTIFICATION
                                )
                            ).play()
                        } catch (_: Exception) {
                        }

                        try {
                            triggerSystemNotification(
                                context,
                                newNotif.title,
                                newNotif.message,
                                newNotif.actionTarget
                            )
                        } catch (_: Exception) {
                        }
                    }
                } else { globalNotifCount = 0 }
                isGlobalInitial = false
            }

        var isUserInitial = true

        val userNotifListener = uid?.let { currentUid ->
            firestoreDb.collection("Users")
                .document(currentUid)
                .collection("UserNotifications")
                .orderBy("timestamp", Query.Direction.DESCENDING)
                .addSnapshotListener { snapshot, error ->
                    if (error == null && snapshot != null) {
                        userNotifCount = snapshot.documents.size

                        val hasChanges = snapshot.documentChanges.any {
                            it.type == DocumentChange.Type.ADDED
                        }

                        if (!snapshot.isEmpty) {
                            val newNotif =
                                snapshot.documents[0].toObject(NotificationModel::class.java)

                            if (newNotif != null &&
                                !isUserInitial &&
                                hasChanges &&
                                !popupVisible
                            ) {
                                latestAppNotif = newNotif
                                popupVisible = true
                                isFlyingToBell = false

                                try {
                                    RingtoneManager.getRingtone(
                                        context,
                                        RingtoneManager.getDefaultUri(
                                            RingtoneManager.TYPE_NOTIFICATION
                                        )
                                    ).play()
                                } catch (_: Exception) {
                                }

                                try {
                                    triggerSystemNotification(
                                        context,
                                        newNotif.title,
                                        newNotif.message,
                                        newNotif.actionTarget
                                    )
                                } catch (_: Exception) {
                                }
                            }
                        }
                    } else { userNotifCount = 0 }
                    isUserInitial = false
                }
        }

        onDispose {
            userListener?.remove()
            bannerListener.remove()
            globalNotifListener.remove()
            userNotifListener?.remove()
        }
    }

    LaunchedEffect(uid) {
        if (uid != null) {
            FirebaseFirestore.getInstance()
                .collection("Users")
                .document(uid)
                .update("lastActive", System.currentTimeMillis())
        }
    }

    // Existing banner auto-slide behavior preserved, with smoother animation.
    val bannerPagerState = rememberPagerState(
        pageCount = { if (banners.isEmpty()) 1 else banners.size }
    )

    LaunchedEffect(bannerPagerState, banners.size) {
        if (banners.size > 1) {
            while (true) {
                delay(4000)

                if (!bannerPagerState.isScrollInProgress) {
                    val nextPage =
                        (bannerPagerState.currentPage + 1) % banners.size

                    bannerPagerState.animateScrollToPage(
                        page = nextPage,
                        animationSpec = tween(
                            durationMillis = 900,
                            easing = FastOutSlowInEasing
                        )
                    )
                }
            }
        }
    }

    // 🔥 FIX: Removed all '\n' to ensure FULL NAMES are displayed 🔥
    val gridItemsList = listOf(
        HomeGridItem(
            name = "Paid Courses",
            subtitle = "Learn from experts",
            iconVector = Icons.Default.School,
            color = Color.White,
            cardStart = Color(0xFFE1F0FF),
            cardEnd = Color(0xFFB9DDFF)
        ),
        HomeGridItem(
            name = "Free Courses",
            subtitle = "Quality study material",
            iconVector = Icons.Default.OndemandVideo,
            color = Color.White,
            cardStart = Color(0xFFDFFAF0),
            cardEnd = Color(0xFFBCEFE0)
        ),
        HomeGridItem(
            name = "Job Alerts",
            subtitle = "Latest notifications",
            iconVector = Icons.Default.NotificationsActive,
            color = Color.White,
            cardStart = Color(0xFFFFE9D9),
            cardEnd = Color(0xFFFFD0B5)
        ),
        HomeGridItem(
            name = "Latest Result",
            subtitle = "Check your results",
            iconRes = R.drawable.result,
            color = Color.White,
            cardStart = Color(0xFFFFDCE9),
            cardEnd = Color(0xFFFFBFD5)
        ),
        HomeGridItem(
            name = "Documents",
            subtitle = "Syllabus, PDF, etc.",
            iconRes = R.drawable.doc,
            color = Color.White,
            cardStart = Color(0xFFE9E0FF),
            cardEnd = Color(0xFFD4C6FF)
        ),
        HomeGridItem(
            name = "Books",
            subtitle = "Read & prepare",
            iconVector = Icons.Default.MenuBook,
            color = Color.White,
            cardStart = Color(0xFFE0F0FF),
            cardEnd = Color(0xFFBBDFFF)
        ),
        HomeGridItem(
            name = "Online Exam (CBT)",
            subtitle = "Practice & improve",
            iconVector = Icons.Default.Computer,
            color = Color.White,
            cardStart = Color(0xFFFFE0E0),
            cardEnd = Color(0xFFFFC8C8)
        ),
        HomeGridItem(
            name = "Age Calculator",
            subtitle = "Calculate your age",
            iconVector = Icons.Default.Calculate,
            color = Color.White,
            cardStart = Color(0xFFDFF9EE),
            cardEnd = Color(0xFFBDEEDB)
        ),
        HomeGridItem(
            name = "Previous Year",
            subtitle = "Access past papers",
            iconVector = Icons.Default.Update,
            color = Color.White,
            cardStart = Color(0xFFFFF2C9),
            cardEnd = Color(0xFFFFE39A)
        ),
        HomeGridItem(
            name = "Syllabus",
            subtitle = "View complete syllabus",
            iconVector = Icons.Default.Apps,
            color = Color.White,
            cardStart = Color(0xFFE8E0FF),
            cardEnd = Color(0xFFD5C8FF)
        )
    )

    Box(modifier = Modifier.fillMaxSize()) {
        ModalNavigationDrawer(
            drawerState = drawerState,
            gesturesEnabled = !showSearchOverlay,
            drawerContent = {
                SidebarContent(
                    auth,
                    navController,
                    selectedTab,
                    { selectedTab = it },
                    { coroutineScope.launch { drawerState.close() } }
                )
            }
        ) {
            Scaffold(
                containerColor = Color.Transparent,

                topBar = {
                    PremiumHomeTopBar(
                        userName = userName,
                        totalActiveNotifs = globalNotifCount + userNotifCount,
                        onMenuClick = {
                            coroutineScope.launch { drawerState.open() }
                        },
                        onNotificationClick = {
                            navController.navigate("notification_screen") {
                                launchSingleTop = true
                            }
                        },
                        onSearchClick = {
                            showSearchOverlay = true
                        }
                    )
                },

                bottomBar = {
                    PremiumBottomBar(
                        selectedTab = selectedTab,
                        hasNotification = (globalNotifCount + userNotifCount) > 0,
                        onTabSelected = { selectedTab = it }
                    )
                }
            ) { padding ->
                Column(
                    modifier = Modifier
                        .fillMaxSize()
                        .padding(padding)
                        .background(
                            Brush.verticalGradient(
                                colors = listOf(
                                    Color(0xFFF5FAFF),
                                    Color(0xFFEAF4FF)
                                )
                            )
                        )
                ) {
                    when (selectedTab) {
                        "Home" -> {
                            HomeDashboard(
                                userName = userName,
                                banners = banners,
                                isBannerLoading = isBannerLoading,
                                bannerPagerState = bannerPagerState,
                                gridItemsList = gridItemsList,
                                navController = navController,
                                onSearchClick = { showSearchOverlay = true }
                            )
                        }

                        "Updates" -> {
                            UserUpdateScreen(navController)
                        }

                        "Downloads" -> {
                            DownloadsSection(
                                navController,
                                auth,
                                title = "Your Downloads",
                                emptyMessage = "No Downloads Available.",
                                isPurchasesTab = false
                            )
                        }

                        "Purchases" -> {
                            DownloadsSection(
                                navController = navController,
                                auth = auth,
                                title = "Your Purchases",
                                emptyMessage = "No Purchases Right Now.",
                                isPurchasesTab = true,

                                    selectedTab = "Downloads"
                                }
                            )
                        }
                    }
                }
            }
        }

        // Existing search overlay preserved.
        AnimatedVisibility(
            visible = showSearchOverlay,
            enter = fadeIn(animationSpec = tween(300)),
            exit = fadeOut(animationSpec = tween(300))
        ) {
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .background(Color.Black.copy(alpha = 0.6f))
                    .pointerInput(Unit) {
                        detectTapGestures { }
                    }
                    .zIndex(200f)
            ) {
                SearchBoxScreen(
                    navController = navController,
                    onClose = { showSearchOverlay = false }
                )
            }
        }

        // Existing notification popup/animation preserved.
        if (popupVisible && latestAppNotif != null) {
            val scale by animateFloatAsState(
                targetValue = if (isFlyingToBell) 0.05f else 1f,
                animationSpec = tween(
                    durationMillis = 700,
                    easing = FastOutSlowInEasing
                ),
                label = "notification_scale"
            )

            val translationX by animateFloatAsState(
                targetValue = if (isFlyingToBell) 420f else 0f,
                animationSpec = tween(
                    durationMillis = 700,
                    easing = FastOutSlowInEasing
                ),
                label = "notification_translation_x"
            )

            val translationY by animateFloatAsState(
                targetValue = if (isFlyingToBell) -950f else 0f,

                    durationMillis = 700,
                    easing = FastOutSlowInEasing

                    }
                }
            }
        }

        }

            pageCount = { menuPages.size }

                                            "Free Courses" ->
                                                navController.navigate("free_courses_list") { launchSingleTop = true }
                                            "Job Alerts" ->
                                                navController.navigate("job_alerts") { launchSingleTop = true }

                                                navController.navigate("document_screen") { launchSingleTop = true }
                                            "Latest Result" ->

                                            "Books" ->
                                                navController.navigate("book_category_screen") { launchSingleTop = true }
                                            "Previous Year" ->
                                                navController.navigate("py_category_screen") { launchSingleTop = true }
                                            "Online Exam (CBT)" ->
                                                navController.navigate("online_cbt_exam") { launchSingleTop = true }
                                            "Syllabus" ->
                                                navController.navigate("user_syllabus_categories") { launchSingleTop = true }

@Composable
private fun SearchPill(
    modifier: Modifier = Modifier,
    onClick: () -> Unit,
    enabled: Boolean = true
) {
    val interactionSource = remember { MutableInteractionSource() }
    val pressed by interactionSource.collectIsPressedAsState()
    val hovered by interactionSource.collectIsHoveredAsState()

    val scale by animateFloatAsState(
        targetValue = when {
            pressed -> 0.985f
            hovered -> 1.01f
            else -> 1f
        },
        animationSpec = tween(120),
        label = "search_press_scale"
    )

    // 🔥 FIX: SEARCH BOX REDUCED HEIGHT 🔥
    Box(
        modifier = modifier
            .height(40.dp) // Compact reference search height
            .graphicsLayer {
                scaleX = scale
                scaleY = scale
            }
            .shadow(
                elevation = 5.dp,
                shape = RoundedCornerShape(30.dp)
            )
            .clip(RoundedCornerShape(30.dp))
            .background(Color.White)
            .hoverable(interactionSource)
            .clickable(
                enabled = enabled,
                interactionSource = interactionSource,
                indication = null,
                onClick = onClick
            )
            .padding(horizontal = 14.dp),
        contentAlignment = Alignment.CenterStart
    ) {
        Icon(
            imageVector = Icons.Default.Search,
            contentDescription = "Search",
            tint = Color(0xFF0C2460),
            modifier = Modifier.size(18.dp)
        )

        Text(
            text = "Search courses, exams...",
            modifier = Modifier.padding(start = 8.dp, end = 26.dp),
            color = Color(0xFF61749A),
            fontSize = 9.sp, // Compact reference size
            fontFamily = Poppins,
            fontWeight = FontWeight.Medium,
            maxLines = 1,
            overflow = TextOverflow.Ellipsis
        )

    }
}

@Composable
private fun HomeMenuCard(
    item: HomeGridItem,
    modifier: Modifier = Modifier,
    onClick: () -> Unit
) {
    val interactionSource = remember { MutableInteractionSource() }
    val pressed by interactionSource.collectIsPressedAsState()
    val hovered by interactionSource.collectIsHoveredAsState()

    val scale by animateFloatAsState(
        targetValue = when {
            pressed -> 0.965f
            hovered -> 1.015f
            else -> 1f
        },
        animationSpec = tween(
            durationMillis = 140,
            easing = FastOutSlowInEasing
        ),
        label = "menu_card_scale"
    )

    val elevation by animateDpAsState(
        targetValue = if (pressed) 1.dp else 5.dp,
        animationSpec = tween(140),
        label = "menu_card_elevation"
    )

    val arrowScale by animateFloatAsState(
        targetValue = if (pressed) 0.90f else 1f,
        animationSpec = tween(140),
        label = "menu_arrow_scale"
    )

    Box(
        modifier = modifier
            // Removed fixed height, it expands naturally inside the Row
            .graphicsLayer {
                scaleX = scale
                scaleY = scale
            }
            .shadow(
                elevation = elevation,
                shape = RoundedCornerShape(20.dp)
            )
            .clip(RoundedCornerShape(20.dp))
            .hoverable(interactionSource)
            .background(
                Brush.linearGradient(
                    colors = listOf(
                        item.cardStart,
                        item.cardEnd
                    )
                )
            )
            .clickable(
                interactionSource = interactionSource,
                indication = null,
                onClick = onClick
            )
            .padding(10.dp) // Uniform padding
    ) {
        // Decorative translucent circles dynamically positioned.
        Box(
            modifier = Modifier
                .size(60.dp)
                .align(Alignment.TopEnd)
                .offset(x = 20.dp, y = (-20).dp)

                .background(Color.White.copy(alpha = 0.16f))

        Box(

                .offset(x = 10.dp, y = 10.dp)
                .clip(CircleShape)
                .background(Color.White.copy(alpha = 0.13f))
        )

        // Icon circle.
        Box(

                .size(50.dp) // Reference-sized menu icon circle

                    elevation = 4.dp,
                    shape = CircleShape
                )

            contentAlignment = Alignment.Center

                    imageVector = item.iconVector,

                    modifier = Modifier.size(28.dp)

                    painter = painterResource(id = item.iconRes),
